F.X — 第F章总结与测验

● 机器译文 课程 F.X 查看英文原文

一个 constexpr 函数是允许在常量表达式中调用的函数。要使函数成为 constexpr 函数，我们只需在返回类型前使用 constexpr 关键字。Constexpr 函数仅在用于需要常量表达式的上下文时保证在编译时求值。否则，它们可能在编译时（如果符合条件）或运行时求值。Constexpr 函数是隐式内联的，编译器必须看到 constexpr 函数的完整定义才能在编译时调用它。

一个 consteval 函数 是必须在编译时求值的函数。Consteval 函数在其他方面遵循与 constexpr 函数相同的规则。

小测验时间

问题 #1

添加 const 和/或 constexpr 到以下程序：

#include <iostream>

// 从用户获取塔高并返回
double getTowerHeight()
{
	std::cout << "Enter the height of the tower in meters: ";
	double towerHeight{};
	std::cin >> towerHeight;
	return towerHeight;
}

// 返回经过“seconds”秒后球离地面的高度
double calculateBallHeight(double towerHeight, int seconds)
{
	double gravity{ 9.8 };

	// 使用公式：[ s = u * t + (a * t^2) / 2 ]，这里 u（初速度）= 0
	double distanceFallen{ (gravity * (seconds * seconds)) / 2.0 };
	double currentHeight{ towerHeight - distanceFallen };

	return currentHeight;
}

// 打印球离地高度
void printBallHeight(double ballHeight, int seconds)
{
	if (ballHeight > 0.0)
		std::cout << "At " << seconds << " seconds, the ball is at height: " << ballHeight << " meters\n";
	else
		std::cout << "At " << seconds << " seconds, the ball is on the ground.\n";
}

// 计算当前球高并打印
// 这是一个辅助函数，使操作更简单
void printCalculatedBallHeight(double towerHeight, int seconds)
{
	double ballHeight{ calculateBallHeight(towerHeight, seconds) };
	printBallHeight(ballHeight, seconds);
}

int main()
{
	double towerHeight{ getTowerHeight() };

	printCalculatedBallHeight(towerHeight, 0);
	printCalculatedBallHeight(towerHeight, 1);
	printCalculatedBallHeight(towerHeight, 2);
	printCalculatedBallHeight(towerHeight, 3);
	printCalculatedBallHeight(towerHeight, 4);
	printCalculatedBallHeight(towerHeight, 5);

	return 0;
}

显示答案

#include <iostream>

// 此函数不应设为 constexpr，因为输出和输入只能在运行时进行。
// 执行输出和输入的 `operator<<` 和 `operator>>` 版本不支持 constexpr。
double getTowerHeight()
{
	std::cout << "Enter the height of the tower in meters: ";
	double towerHeight{};
	std::cin >> towerHeight;
	return towerHeight;
}

// 返回经过“seconds”秒后离地面的高度
// 此函数设为 constexpr，因为它仅根据输入计算一个值并返回。
// 算术运算可以在编译时完成，并且没有调用非 constexpr 函数。
// 提醒：constexpr 函数可以在编译时或运行时求值。
//   如果其参数是 constexpr，则可以在编译时调用。
//   在这种情况下，它在运行时被调用，因为 towerHeight 的参数不是 constexpr。
// 如果一个函数可以设为 constexpr，就应该设为 constexpr。
// 记住：函数参数不是 constexpr，即使在 constexpr 函数中也是如此
constexpr double calculateBallHeight(double towerHeight, int seconds)
{
	constexpr double gravity{ 9.8 }; // constexpr，因为它是编译时常量

	// 使用公式：[ s = u * t + (a * t^2) / 2 ]，这里 u（初速度）= 0
	// 这些变量不能是 constexpr，因为它们的初始化器不是常量表达式
	const double distanceFallen{ (gravity * (seconds * seconds)) / 2.0 };
	const double currentHeight{ towerHeight - distanceFallen };

	return currentHeight;
}

// 此函数不应设为 constexpr，因为输出和输入只能在运行时进行。
// 执行输出和输入的 `operator<<` 和 `operator>>` 版本不支持 constexpr。
void printBallHeight(double ballHeight, int seconds)
{
	if (ballHeight > 0.0)
		std::cout << "At " << seconds << " seconds, the ball is at height: " << ballHeight << " meters\n";
	else
		std::cout << "At " << seconds << " seconds, the ball is on the ground.\n";
}

// 此函数不应设为 constexpr，因为输出和输入只能在运行时进行。
// 执行输出和输入的 `operator<<` 和 `operator>>` 版本不支持 constexpr。
void printCalculatedBallHeight(double towerHeight, int seconds)
{
	// height 只能是 const（不能是 constexpr），因为它的初始化器不是常量表达式
	const double ballHeight{ calculateBallHeight(towerHeight, seconds) };
	printBallHeight(ballHeight, seconds);
}

int main()
{
	// towerHeight 只能是 const（不能是 constexpr），因为它的初始化器不是常量表达式
	const double towerHeight{ getTowerHeight() };

	printCalculatedBallHeight(towerHeight, 0);
	printCalculatedBallHeight(towerHeight, 1);
	printCalculatedBallHeight(towerHeight, 2);
	printCalculatedBallHeight(towerHeight, 3);
	printCalculatedBallHeight(towerHeight, 4);
	printCalculatedBallHeight(towerHeight, 5);

	return 0;
}
