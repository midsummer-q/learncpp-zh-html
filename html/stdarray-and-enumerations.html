17.6 — std::array 与枚举

● 机器译文 课程 17.6 查看英文原文

在课程 16.9 -- 使用枚举器进行数组索引和长度，我们讨论了数组和枚举。

现在我们已经有了 constexpr std::array 在我们的工具箱中，我们将继续讨论并展示一些额外的技巧。

使用 static_assert 确保数组初始化器的正确数量

当用元素值初始化一个 constexpr std::array 使用 CTAD，编译器会根据初始化器的数量推断数组的长度。如果提供的初始化器少于应有的数量，数组将比预期的短，索引它可能导致未定义行为。

例如：

#include <array>
#include <iostream>

enum StudentNames
{
    kenny, // 0
    kyle, // 1
    stan, // 2
    butters, // 3
    cartman, // 4
    max_students // 5
};

int main()
{
    constexpr std::array testScores { 78, 94, 66, 77 }; // 哎呀，只有 4 个值

    std::cout << "Cartman got a score of " << testScores[StudentNames::cartman] << '\n'; // 由于无效索引导致的未定义行为

    return 0;
}

每当可以合理检查 constexpr std::array 中初始化器的数量时，你可以使用 static_assert 进行检查：

#include <array>
#include <iostream>

enum StudentNames
{
    kenny, // 0
    kyle, // 1
    stan, // 2
    butters, // 3
    cartman, // 4
    max_students // 5
};

int main()
{
    constexpr std::array testScores { 78, 94, 66, 77 };

    // 确保测试分数的数量与学生数量相同
    static_assert(std::size(testScores) == max_students); // 编译错误：static_assert 条件失败

    std::cout << "Cartman got a score of " << testScores[StudentNames::cartman] << '\n';

    return 0;
}

这样，如果你以后添加了一个新的枚举器但忘记在 testScores中添加相应的初始化器，程序将无法编译。

你也可以使用 static_assert 来确保两个不同的 constexpr std::array 具有相同的长度。

使用 constexpr 数组实现更好的枚举输入和输出

在课程 13.5 -- 重载 I/O 运算符简介，我们介绍了几种输入和输出枚举器名称的方法。为了辅助这项任务，我们编写了辅助函数，将枚举器转换为字符串，反之亦然。这些函数各自拥有自己的（重复的）字符串字面量集合，并且我们必须专门编写逻辑来检查每一个：

constexpr std::string_view getPetName(Pet pet)
{
    switch (pet)
    {
    case cat:   return "cat";
    case dog:   return "dog";
    case pig:   return "pig";
    case whale: return "whale";
    default:    return "???";
    }
}

constexpr std::optional<Pet> getPetFromString(std::string_view sv)
{
    if (sv == "cat")   return cat;
    if (sv == "dog")   return dog;
    if (sv == "pig")   return pig;
    if (sv == "whale") return whale;

    return {};
}

这意味着如果我们添加一个新的枚举器，我们必须记得更新这些函数。

让我们稍微改进这些函数。如果我们的枚举器的值从 0 开始并且顺序递增（大多数枚举都是如此），我们可以使用一个数组来存储每个枚举器的名称。

这使我们能够做两件事：

  1. 使用枚举器的值作为索引来获取该枚举器的名称。
  2. 使用循环遍历所有名称，并能够根据索引将名称与枚举器关联起来。
#include <array>
#include <iostream>
#include <string>
#include <string_view>

namespace Color
{
    enum Type
    {
        black,
        red,
        blue,
        max_colors
    };

    // 使用 sv 后缀，以便 std::array 推断类型为 std::string_view
    using namespace std::string_view_literals; // 用于 sv 后缀
    constexpr std::array colorName { "black"sv, "red"sv, "blue"sv };

    // 确保我们为所有颜色定义了字符串
    static_assert(std::size(colorName) == max_colors);
};

constexpr std::string_view getColorName(Color::Type color)
{
    // 我们可以使用枚举器索引数组来获取枚举器的名称
    return Color::colorName[static_cast<std::size_t>(color)];
}

// 教 operator<< 如何打印 Color
// std::ostream 是 std::cout 的类型
// 返回类型和参数类型是引用（以防止复制）！
std::ostream& operator<<(std::ostream& out, Color::Type color)
{
    return out << getColorName(color);
}

// 教导 operator>> 如何按名称输入 Color
// 我们通过非常量引用传递 color，以便函数可以修改其值
std::istream& operator>> (std::istream& in, Color::Type& color)
{
    std::string input {};
    std::getline(in >> std::ws, input);

    // 遍历名称列表，看看是否能找到匹配的名称
    for (std::size_t index=0; index < Color::colorName.size(); ++index)
    {
        if (input == Color::colorName[index])
        {
            // 如果找到匹配的名称，我们可以根据其索引获取枚举器值
            color = static_cast<Color::Type>(index);
            return in;
        }
    }

    // 没有找到匹配，所以输入无效
    // 因此我们将输入流设置为失败状态
    in.setstate(std::ios_base::failbit);

    // 提取失败时，operator>> 将基本类型零初始化
    // 取消注释以下行以使此运算符执行相同操作
    // color = {};
    return in;
}

int main()
{
    auto shirt{ Color::blue };
    std::cout << "Your shirt is " << shirt << '\n';

    std::cout << "Enter a new color: ";
    std::cin >> shirt;
    if (!std::cin)
        std::cout << "Invalid\n";
    else
        std::cout << "Your shirt is now " << shirt << '\n';

    return 0;
}

这会打印：

Your shirt is blue
Enter a new color: red
Your shirt is now red

基于范围的 for 循环与枚举

有时我们会遇到需要遍历枚举的枚举器的情况。虽然我们可以使用整数索引的 for 循环来实现，但这可能需要大量地将整数索引静态转换为枚举类型。

#include <array>
#include <iostream>
#include <string_view>

namespace Color
{
    enum Type
    {
        black,
        red,
        blue,
        max_colors
    };

    // 使用 sv 后缀，以便 std::array 推断类型为 std::string_view
    using namespace std::string_view_literals; // 用于 sv 后缀
    constexpr std::array colorName { "black"sv, "red"sv, "blue"sv };

    // 确保我们为所有颜色定义了字符串
    static_assert(std::size(colorName) == max_colors);
};

constexpr std::string_view getColorName(Color::Type color)
{
    return Color::colorName[color];
}

// 教 operator<< 如何打印 Color
// std::ostream 是 std::cout 的类型
// 返回类型和参数类型是引用（以防止复制）！
std::ostream& operator<<(std::ostream& out, Color::Type color)
{
    return out << getColorName(color);
}

int main()
{
    // 使用 for 循环遍历所有颜色
    for (int i=0; i < Color::max_colors; ++i )
        std::cout << static_cast<Color::Type>(i) << '\n';

    return 0;
}

不幸的是，基于范围的 for 循环不允许你遍历枚举的枚举器：

#include <array>
#include <iostream>
#include <string_view>

namespace Color
{
    enum Type
    {
        black,
        red,
        blue,
        max_colors
    };

    // 使用 sv 后缀，以便 std::array 推断类型为 std::string_view
    using namespace std::string_view_literals; // 用于 sv 后缀
    constexpr std::array colorName { "black"sv, "red"sv, "blue"sv };

    // 确保我们为所有颜色定义了字符串
    static_assert(std::size(colorName) == max_colors);
};

constexpr std::string_view getColorName(Color::Type color)
{
    return Color::colorName[color];
}

// 教 operator<< 如何打印 Color
// std::ostream 是 std::cout 的类型
// 返回类型和参数类型是引用（以防止复制）！
std::ostream& operator<<(std::ostream& out, Color::Type color)
{
    return out << getColorName(color);
}

int main()
{
    for (auto c: Color::Type) // 编译错误：无法遍历枚举
        std::cout << c < '\n';

    return 0;
}

有许多创造性的解决方案。由于我们可以在数组上使用基于范围的 for 循环，最直接的解决方案之一是创建一个 constexpr std::array 包含每个枚举器，然后遍历它。这种方法仅在枚举器具有唯一值时有效。

#include <array>
#include <iostream>
#include <string_view>

namespace Color
{
    enum Type
    {
        black,     // 0
        red,       // 1
        blue,      // 2
        max_colors // 3
    };

    using namespace std::string_view_literals; // 用于 sv 后缀
    constexpr std::array colorName { "black"sv, "red"sv, "blue"sv };
    static_assert(std::size(colorName) == max_colors);

    constexpr std::array types { black, red, blue }; // 包含所有枚举器的 std::array
    static_assert(std::size(types) == max_colors);
};

constexpr std::string_view getColorName(Color::Type color)
{
    return Color::colorName[color];
}

// 教 operator<< 如何打印 Color
// std::ostream 是 std::cout 的类型
// 返回类型和参数类型是引用（以防止复制）！
std::ostream& operator<<(std::ostream& out, Color::Type color)
{
    return out << getColorName(color);
}

int main()
{
    for (auto c: Color::types) // ok：我们可以在 std::array 上使用基于范围的 for
        std::cout << c << '\n';

    return 0;
}

在上面的例子中，由于 Color::types 等于 Color::Type的typedef，变量 c 的元素类型将被推导为 Color::Type，这正是我们想要的！

这会打印：

black
red
blue

小测验时间

定义一个名为 Animal的命名空间。在其中，定义一个包含以下动物的枚举：chicken、dog、cat、elephant、duck 和 snake。同时创建一个名为 Data 的结构体，用于存储每种动物的名称、腿的数量以及它发出的声音。创建一个 std::array 的 Data 数组，并为每种动物填充一个 Data 元素。

要求用户输入一种动物的名称。如果名称与我们的动物名称不匹配，请告知用户。否则，打印该动物的数据。然后打印所有其他与输入不匹配的动物的数据。

例如：

Enter an animal: dog
A dog has 4 legs and says woof.

Here is the data for the rest of the animals:
A chicken has 2 legs and says cluck.
A cat has 4 legs and says meow.
A elephant has 4 legs and says pawoo.
A duck has 2 legs and says quack.
A snake has 0 legs and says hissss.

Enter an animal: frog
That animal couldn't be found.

Here is the data for the rest of the animals:
A chicken has 2 legs and says cluck.
A dog has 4 legs and says woof.
A cat has 4 legs and says meow.
A elephant has 4 legs and says pawoo.
A duck has 2 legs and says quack.
A snake has 0 legs and says hissss.

显示答案

#include <array>
#include <iostream>
#include <string>
#include <string_view>

namespace Animal
{
    enum Type
    {
        chicken,
        dog,
        cat,
        elephant,
        duck,
        snake,
        max_animals
    };

    struct Data
    {
        std::string_view name{};
        int legs{};
        std::string_view sound{};
    };

    constexpr std::array types { chicken, dog, cat, elephant, duck, snake };
    constexpr std::array data {
        Data{ "chicken",    2, "cluck" },
        Data{ "dog",        4, "woof" },
        Data{ "cat",        4, "meow" },
        Data{ "elephant",   4, "pawoo" },
        Data{ "duck",       2, "quack" },
        Data{ "snake",      0, "hissss" },
    };

    static_assert(std::size(types) == max_animals);
    static_assert(std::size(data) == max_animals);
}

// 教导 operator>> 如何按名称输入 Animal
// 我们通过非常量引用传递 animal，以便函数可以修改其值
std::istream& operator>> (std::istream& in, Animal::Type& animal)
{
    std::string input {};
    std::getline(in >> std::ws, input);

    // 看看是否能找到匹配项
    for (std::size_t index=0; index < Animal::data.size(); ++index)
    {
        if (input == Animal::data[index].name)
        {
            animal = static_cast<Animal::Type>(index);
            return in;
        }
    }

    // 没有找到匹配，所以输入无效
    // 因此我们将输入流设置为失败状态
    in.setstate(std::ios_base::failbit);
    return in;
}

void printAnimalData(Animal::Type type)
{
    const Animal::Data& animal { Animal::data[type] };
    std::cout << "A " << animal.name << " has " << animal.legs << " legs and says " << animal.sound << ".\n";    
}

int main()
{
    std::cout << "Enter an animal: ";
    Animal::Type type{};
    std::cin >> type;

    // 如果用户输入不匹配
    if (!std::cin)
    {
        std::cin.clear();
        std::cout << "That animal couldn't be found.\n";
        type = Animal::max_animals; // 设置为无效选项，以便下面不匹配
    }
    else
        printAnimalData(type);


    std::cout << "\nHere is the data for the rest of the animals:\n";
    for (auto a : Animal::types)
    {
        if (a != type)
            printAnimalData(a);
    }

    return 0;
}
